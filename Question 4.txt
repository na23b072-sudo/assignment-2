import numpy as np
import matplotlib.pyplot as plt
from PIL import Image

# --------------------------------------------------
# 1. Load the image
# --------------------------------------------------

image = np.array(Image.open("oceanCapture.jpg"))

# --------------------------------------------------
# 2. Convert RGB image to grayscale
# --------------------------------------------------

gray = np.mean(image, axis=2)

# --------------------------------------------------
# 3. Compute 2D Fourier Transform
# --------------------------------------------------

F = np.fft.fft2(gray)

# --------------------------------------------------
# 4. Calculate magnitude of Fourier coefficients
# --------------------------------------------------

magnitude = np.abs(F)

# Total number of Fourier coefficients
total = F.size

print("Image shape:", gray.shape)
print("Total Fourier components:", total)


# --------------------------------------------------
# Function to reconstruct image
# --------------------------------------------------

def reconstruct(F, percentage):

    # Calculate number of components to retain
    k = int(F.size * percentage)

    # Calculate magnitude
    magnitude = np.abs(F)

    # Find indices of the k largest magnitudes
    indices = np.argpartition(magnitude.ravel(), -k)[-k:]

    # Create an array containing only selected components
    F_compressed = np.zeros_like(F)

    # Flatten both arrays for easy indexing
    F_compressed.ravel()[indices] = F.ravel()[indices]

    # Inverse Fourier Transform
    reconstructed = np.fft.ifft2(F_compressed)

    # Take real part
    reconstructed = np.real(reconstructed)

    return reconstructed


# --------------------------------------------------
# 5. Reconstruct using 10%, 5% and 1%
# --------------------------------------------------

image_10 = reconstruct(F, 0.10)
image_5 = reconstruct(F, 0.05)
image_1 = reconstruct(F, 0.01)


# --------------------------------------------------
# 6. Display results
# --------------------------------------------------

plt.figure(figsize=(15, 8))

plt.subplot(2, 2, 1)
plt.imshow(gray, cmap="gray")
plt.title("Original Image")
plt.axis("off")

plt.subplot(2, 2, 2)
plt.imshow(image_10, cmap="gray")
plt.title("Reconstruction using 10%")
plt.axis("off")

plt.subplot(2, 2, 3)
plt.imshow(image_5, cmap="gray")
plt.title("Reconstruction using 5%")
plt.axis("off")

plt.subplot(2, 2, 4)
plt.imshow(image_1, cmap="gray")
plt.title("Reconstruction using 1%")
plt.axis("off")

plt.tight_layout()
plt.show()
